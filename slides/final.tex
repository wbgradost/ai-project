\begin{frame}{Research question and conditional answer}
  Can AI improve long-run skill while reducing human effort?
  \par\medskip
  \begin{block}{Result in one sentence}
  With $b,\lambda_0,\kappa,\mu>0$ and fixed primitives, AI initially raises
  the stationary high-skill share \alert{if and only if $\eta b>1$},
  where $b$ is effort without AI and $\eta$ measures learning complementarity.
  \end{block}
  The improvement is bounded: assistance eventually eliminates effort,
  leaving only background learning.
  \[
    \lambda(e,a)=\lambda_0+\kappa e(1+\eta a).
  \]
\end{frame}

\begin{frame}{Baseline and contribution}
  \textbf{Aouad, Lykouris, and Zhong (2026), Proposition 3.5:}
  when upward learning rates depend only on effort, more assistance weakly
  shifts stationary skill downward.
  \par\medskip
  \begin{itemize}
    \item Keep the myopic worker problem and effort displacement.
    \item Allow assistance to increase learning from remaining effort.
    \item Derive exact two-state conditions and test simulated dynamics.
  \end{itemize}
  \par\medskip
  The contribution is a stationary-distribution characterization.
  It does not establish a welfare or productivity ranking.
  \source{Baseline: arXiv:2605.11350v1, Prop.~3.5, p.~8.
  Learning complementarity itself is not a novelty claim; see the topic proposal.}
\end{frame}

\begin{frame}{The worker's decision is unchanged}
  For fixed skill $s$ and assistance $a\geq0$,
  \[
    \max_{e\geq0}\;p(s+e+a)-\gamma e,\qquad \gamma>0.
  \]
  $p$ is continuously differentiable, increasing and concave;
  $\limsup_{x\to\infty}p(x)/x<\gamma$ ensures finite maximizers.
  \[
    x^*=\max\arg\max_{x\geq0}\{p(x)-\gamma x\},
    \qquad e^*(s,a)=(x^*-s-a)^+.
  \]
  Retain the original \textbf{largest-maximizer selection} if choices tie.
  \[
    e>0:\ p'(s+e+a)=\gamma;\qquad
    e=0:\ p'(s+a)\leq\gamma.
  \]
  Learning does not enter this objective. Its technology cannot change the FOC.
\end{frame}

\begin{frame}{Two-state skill dynamics}
  \[
    0\leq s_L<s_H,\qquad b=x^*-s_L>0,\qquad
    e_L^*(a)=(b-a)^+.
  \]
  At fixed assistance, the continuous-time Markov chain has
  \[
    Q(a)=
    \begin{pmatrix}
      -\lam(a)&\lam(a)\\ \mu&-\mu
    \end{pmatrix},
    \qquad
    \lam(a)=\lambda_0+\kappa(b-a)^+(1+\eta a).
  \]
  Assume $\lambda_0,\kappa,\mu>0$ and $\eta\geq0$ throughout.
  Balance and normalization give the unique stationary law:
  \[
    (1-\pi_H)\lam=\pi_H\mu
    \quad\Longrightarrow\quad
    \boxed{\pi_H(a)=\frac{\lam(a)}{\lam(a)+\mu}.}
  \]
  Positive rates imply irreducibility. Effort in $H$ does not affect transitions
  because there is no higher state.
\end{frame}

\begin{frame}{Learning complementarity versus effort displacement}
  With positive effort, $0\leq a<b$,
  \[
    \lam(a)=\lambda_0+\kappa(b-a)(1+\eta a).
  \]
  The marginal effect splits into two terms:
  \[
    \lam'(a)=\kappa\left[
      \underbrace{\eta(b-a)}_{\text{learning from remaining effort}}
      -\underbrace{(1+\eta a)}_{\text{lost human effort}}
    \right]
    =\kappa[\eta(b-2a)-1].
  \]
  Complementarity must exceed displacement, rather than merely be positive.
  At $a\geq b$, effort is zero and $\lam(a)=\lambda_0$.
\end{frame}

\begin{frame}{Proposition 1: global stationary skill}
  Under the worker and rate assumptions just stated:
  \par\medskip
  \begin{itemize}
    \item If $\eta b\leq1$, $\pi_H$ strictly decreases on $[0,b]$,
    then is constant. Its unique maximum is at $a=0$.
    \item If $\eta b>1$, define
    \[
      a^*=\frac{b-1/\eta}{2},\qquad a^R=b-\frac1\eta=2a^*.
    \]
    Then $0<a^*<a^R<b$: $\pi_H$ strictly increases on $[0,a^*]$,
    strictly decreases on $[a^*,b]$, and is constant on $[b,\infty)$.
    \item The unique global maximum is $a^*$ in the strong regime.
  \end{itemize}
  At $\eta b=1$, the right derivative at zero is zero, but there is no
  increasing interval.
\end{frame}

\begin{frame}{Three different comparisons}
  In the strong regime $\eta b>1$:
  \par\medskip
  \begin{center}
  \begin{tabular}{ll}
  \toprule
  Question & Exact answer\\
  \midrule
  Positive marginal effect & $0\leq a<a^*$ (right derivative at zero)\\[0.35em]
  Better than no AI & $0<a<a^R$\\[0.35em]
  Better than lower positive AI & depends on both levels\\[0.35em]
  Zero human effort & $a\geq b$\\
  \bottomrule
  \end{tabular}
  \end{center}
  \par\medskip
  For $a^*<a<a^R$, assistance is already reducing stationary skill,
  while skill remains above the no-AI benchmark.
  \[
    \pi_H(a^R)=\pi_H(0),\qquad
    \pi_H(a\geq b)=\frac{\lambda_0}{\lambda_0+\mu}<\pi_H(0).
  \]
\end{frame}

\begin{frame}{Proof: signs, maximum, and the effort boundary}
  On the active-effort region,
  \[
    \pi_H'(a)=\frac{\mu\kappa[\eta(b-2a)-1]}{[\lam(a)+\mu]^2}.
  \]
  Positivity of $\mu,\kappa$ and the denominator transfers the sign
  to $\eta(b-2a)-1$, which strictly decreases when $\eta>0$.
  \par\medskip
  \begin{itemize}
    \item Its root is interior exactly when $\eta b>1$.
    \item Monotonicity plus the constant tail establishes the global maximum.
    \item At $b$, continuity holds, but differentiability fails:
  \end{itemize}
  \[
    \pi_H'(b-)=-\frac{\mu\kappa(1+\eta b)}{(\lambda_0+\mu)^2}<0,
    \qquad \pi_H'(b+)=0.
  \]
  The active branch is strictly concave; the full curve is not globally concave.
\end{frame}

\begin{frame}{Proposition 2: comparing two assistance levels}
  For $0\leq a_\ell<a_h<b$, under the same assumptions,
  \[
  \pi_H(a_h)-\pi_H(a_\ell)=
  \frac{\mu\kappa(a_h-a_\ell)[\eta(b-a_h-a_\ell)-1]}
       {[\lam(a_h)+\mu][\lam(a_\ell)+\mu]}.
  \]
  Therefore
  \[
    \boxed{\pi_H(a_h)>\pi_H(a_\ell)
    \ \Longleftrightarrow\
    \eta(b-a_h-a_\ell)>1.}
  \]
  In the strong regime this is $a_\ell+a_h<a^R$.
  Set $a_\ell=0$ to obtain the no-AI return threshold.
  \par\medskip
  If $a_\ell<b\leq a_h$, skill strictly falls; if $b\leq a_\ell<a_h$,
  the shares are equal. In two states, share ordering is equivalent to FOSD.
\end{frame}

\begin{frame}{Stochastic experiment: what is actually simulated}
  For each fixed $a$, simulate the CTMC with exponential holding times of means
  $1/\lam$ in $L$ and $1/\mu$ in $H$.
  \[
    \widehat A_T=\frac1T\int_0^T{\bf1}_{\{S_t=H\}}\,dt .
  \]
  \begin{itemize}
    \item \MCReplications{} independent paths per cell; seed \MCSeed.
    \item Start in $L$; horizons $T=25,100,400$, with no burn-in.
    \item Include the last partial holding interval; no time discretization.
    \item Estimate standard errors across independent path occupancies.
  \end{itemize}
  A count of visits would mislead: the two-state jump chain alternates.
  We measure the time spent in each state.
\end{frame}

\begin{frame}{Finite horizons: bias and Monte Carlo error}
  Write $r=\lam+\mu$ and start in $L$. Exact predictions are
  \[
    \Pr(S_t=H)=\pi_H(1-e^{-rt}),\qquad
    E[\widehat A_T]=\pi_H\left[1-\frac{1-e^{-rT}}{rT}\right].
  \]
  \[
  \underbrace{\overline A_T-\pi_H}_{\text{gap from stationarity}}
  =
  \underbrace{\overline A_T-E[\widehat A_T]}_{\text{Monte Carlo error}}
  +\underbrace{E[\widehat A_T]-\pi_H}_{\text{initial-state bias}}.
  \]
  We check means against the finite-horizon prediction using exact
  occupation variance, and report nominal $95\%$ intervals from empirical SEs.
  The automated tolerance is six exact SEs; it is a diagnostic, not proof.
\end{frame}

\begin{frame}{What the computational evidence establishes}
  \begin{itemize}
    \item Symbolic factorization checks derivatives, comparisons, and variance.
    \item Independent effort optimization checks the unchanged worker choice.
    \item Numerical checks cover equality, effort boundaries, arbitrary pairs,
    parameter sensitivity, and stationary balance.
    \item CTMC experiments test occupation against exact dynamics;
    increasing $T$ reduces initial-state bias and sampling error in scale.
  \end{itemize}
  \par\medskip
  These are illustrative model checks. They do not estimate complementarity,
  prove the theorem, or identify effects in data.
  \source{Single entry point: python code/verify.py. Outputs, seeds, tolerances, and environment are recorded in code/output/.}
\end{frame}

\begin{frame}{Proof progress and remaining scope}
  \textbf{Completed mathematically:} effort selection, unique stationarity,
  transient probabilities, global shape, exact two-level comparisons,
  thresholds, and rate-scaling invariance.
  \par\medskip
  \textbf{Delicate steps:} preserve tie-breaking under weak concavity;
  handle $\eta=0$, $\eta b=1$, and the kink at $b$;
  distinguish a share level from a gain.
  \par\medskip
  \textbf{Open research scope:} endogenous assistance, forward-looking effort,
  empirical identification, and more than two skill states.
  No result for a general $N$-state chain is claimed.
\end{frame}
