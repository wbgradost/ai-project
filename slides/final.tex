\begin{frame}{Pregunta de investigación y contribución}
  ¿Puede la IA mejorar las habilidades a largo plazo mientras reduce el esfuerzo humano?
  \par\medskip
  \begin{block}{Resultado central, con su condición}
  En el modelo de dos estados, con $b,\lambda_0,\kappa,\mu>0$ y $\eta\geq0$,
  la asistencia aumenta inicialmente la proporción estacionaria de habilidad alta
  \alert{si y solo si $\eta b>1$}, aunque desplaza esfuerzo humano.
  \end{block}
  \[
    \lambda(e,a)=\lambda_0+\kappa e(1+\eta a).
  \]
  $b$ es el esfuerzo sin IA; $\eta$ mide la complementariedad entre IA y esfuerzo
  en el aprendizaje. La mejora es acotada: cuando desaparece el esfuerzo,
  solo permanece el aprendizaje basal.
\end{frame}

\begin{frame}{Del desplazamiento del esfuerzo a nuestra extensión}
  \textbf{Aouad, Lykouris y Zhong (2026), Proposición 3.5:}
  si el aprendizaje depende solo del esfuerzo, una mayor asistencia desplaza
  débilmente hacia abajo la distribución estacionaria de habilidades.
  \par\medskip
  \begin{itemize}
    \item Conservamos el problema miope del trabajador y su regla de desempate.
    \item Modificamos únicamente la tecnología de aprendizaje: la IA puede
    elevar el rendimiento del esfuerzo que permanece.
    \item Caracterizamos exactamente los umbrales y las comparaciones en dos estados.
  \end{itemize}
  \par\medskip
  La contribución describe la distribución de habilidades; no establece un
  ordenamiento de bienestar ni una política óptima de asistencia.
  \source{Aouad, Lykouris y Zhong, \textit{Human-AI Productivity Paradoxes},
  arXiv:2605.11350v1, Prop.~3.5. Extensión del manuscrito propio fijado en 7cc3a740.}
\end{frame}

\begin{frame}{Lema 1: la decisión del trabajador no cambia}
  Para habilidad fija $s\geq0$ y asistencia $a\geq0$,
  \[
    \max_{e\geq0}\;p(s+e+a)-\gamma e,\qquad \gamma>0.
  \]
  $p$ es continuamente diferenciable, no decreciente y cóncava;
  $\limsup_{x\to\infty}p(x)/x<\gamma$ garantiza máximos finitos.
  \[
    x^*=\max\arg\max_{x\geq0}\{p(x)-\gamma x\},\qquad
    e^*(s,a)=\max\{x^*-s-a,0\}.
  \]
  Se elige el \textbf{mayor maximizador} en caso de empate.
  \[
    e>0:\ p'(s+e+a)=\gamma;\qquad
    e=0:\ p'(s+a)\leq\gamma.
  \]
  El aprendizaje no entra en el objetivo miope: su modificación conserva
  la condición de primer orden y el desplazamiento del esfuerzo.
\end{frame}

\begin{frame}{Lema 2: transiciones y distribución estacionaria}
  \[
    0\leq s_L<s_H,\quad b=x^*-s_L>0,\qquad
    \boxed{e_L^*(a)=\max\{b-a,0\}.}
  \]
  Para cada asistencia fija $a\geq0$, la cadena de Markov de tiempo continuo tiene
  \[
    Q(a)=\begin{pmatrix}-\lam(a)&\lam(a)\\ \mu&-\mu\end{pmatrix},\quad
    \lam(a)=\lambda_0+\kappa\max\{b-a,0\}(1+\eta a).
  \]
  Suponemos $\lambda_0,\kappa,\mu>0$ y $\eta\geq0$ en todo el análisis.
  El balance y la normalización implican una distribución estacionaria única:
  \[
    (1-\pi_H)\lam=\pi_H\mu
    \quad\Longrightarrow\quad
    \boxed{\pi_H(a)=\frac{\lam(a)}{\lam(a)+\mu}.}
  \]
  Las tasas positivas garantizan irreducibilidad. El esfuerzo en $H$ no
  afecta las transiciones porque no existe un estado superior.
\end{frame}

\begin{frame}{Dos fuerzas contrapuestas en el aprendizaje}
  En la región con esfuerzo positivo, $0\leq a<b$, definimos
  \[
    L(a)=\lambda_0+\kappa(b-a)(1+\eta a)=\lam(a).
  \]
  Su efecto marginal se descompone en
  \[
    L'(a)=\kappa\left[
      \underbrace{\eta(b-a)}_{\text{rendimiento del esfuerzo restante}}
      -\underbrace{(1+\eta a)}_{\text{esfuerzo desplazado}}
    \right]
    =\kappa[\eta(b-2a)-1].
  \]
  \begin{block}{Mecanismo económico}
  La IA reduce el esfuerzo, pero puede aumentar su productividad para aprender.
  El efecto neto es positivo solamente si la complementariedad supera
  el desplazamiento: $\eta(b-2a)>1$.
  \end{block}
  Para $a\geq b$, $e_L^*=0$ y $\lam=\lambda_0$: la complementariedad ya no opera.
\end{frame}

\begin{frame}{Proposición 1: forma global y máximo estacionario}
  Bajo los supuestos de los dos lemas anteriores:
  \begin{itemize}
    \item Si $\eta b\leq1$, $\pi_H$ disminuye estrictamente en $[0,b]$
    y después es constante. Su máximo global único está en $a=0$.
    \item Si $\eta b>1$, definimos
    \[
      a^*=\frac{b-1/\eta}{2},\qquad a_R=b-\frac1\eta=2a^*.
    \]
    Entonces $0<a^*<a_R<b$: $\pi_H$ aumenta estrictamente en $[0,a^*]$,
    disminuye estrictamente en $[a^*,b]$ y es constante en $[b,\infty)$.
    \item El máximo global único es $a^*$ en el régimen fuerte.
    La mejora frente a no usar IA ocurre exactamente cuando $0<a<a_R$.
  \end{itemize}
  \par\smallskip
  En la frontera $\eta b=1$, $\pi_H'(0+)=0$, pero no existe un intervalo creciente.
  Para $a\geq b$, $\pi_H=\lambda_0/(\lambda_0+\mu)<\pi_H(0)$.
\end{frame}

\begin{frame}{Demostración: signos, curvatura y frontera del esfuerzo}
  Para $0\leq a<b$ (derivada derecha en cero),
  \[
    \boxed{\pi_H'(a)=\frac{\mu\kappa[\eta(b-2a)-1]}{[L(a)+\mu]^2}.}
  \]
  $\mu,\kappa>0$ y el denominador positivo hacen que el signo dependa
  exclusivamente de $\eta(b-2a)-1$. Su raíz es positiva si y solo si $\eta b>1$.
  \par\smallskip
  Si $f(a)=(b-a)(1+\eta a)$, en $(0,b)$:
  \[
    \pi_H''=-\frac{2\mu\kappa\eta}{(L+\mu)^2}
       -\frac{2\mu\kappa^2[f'(a)]^2}{(L+\mu)^3}<0.
  \]
  Si $\eta=0$, $f'=-1$ conserva la desigualdad estricta. En $b$ hay continuidad,
  pero no diferenciabilidad:
  \[
    \pi_H'(b-)=-\frac{\mu\kappa(1+\eta b)}{(\lambda_0+\mu)^2}<0,
    \qquad \pi_H'(b+)=0.
  \]
  La rama activa es estrictamente cóncava; la curva completa no es cóncava.
\end{frame}

\begin{frame}{Proposición 2: comparar dos niveles de asistencia}
  Para $0\leq a_\ell<a_h<b$, bajo los mismos supuestos,
  \[
    \pi_H(a_h)-\pi_H(a_\ell)=
    \frac{\mu\kappa(a_h-a_\ell)[\eta(b-a_h-a_\ell)-1]}
         {[L(a_h)+\mu][L(a_\ell)+\mu]}.
  \]
  \[
    \boxed{\pi_H(a_h)>\pi_H(a_\ell)
      \iff\eta(b-a_h-a_\ell)>1.}
  \]
  \begin{itemize}
    \item Igualdad si y solo si $\eta(b-a_h-a_\ell)=1$; disminución si es menor que uno.
    \item Si $a_\ell<b\leq a_h$, la habilidad disminuye estrictamente.
    Si $b\leq a_\ell<a_h$, las proporciones son iguales.
    \item En dos estados, ordenar $\pi_H$ equivale a ordenar la habilidad media
    y la dominancia estocástica de primer orden.
  \end{itemize}
  La fórmula activa también es válida en $a_h=b$.
\end{frame}

\begin{frame}{Demostración: una factorización para tres comparaciones}
  Restar las fracciones estacionarias elimina el producto de las tasas:
  \[
    \pi_H(v)-\pi_H(u)=
    \frac{\mu[\lam(v)-\lam(u)]}{[\lam(v)+\mu][\lam(u)+\mu]}.
  \]
  Para $0\leq u<v\leq b$, expandir y factorizar da
  \[
    L(v)-L(u)=\kappa(v-u)[\eta(b-u-v)-1].
  \]
  Los demás factores son positivos. Así obtenemos condiciones necesarias
  y suficientes, sin asumir el signo que se pretende demostrar.
  \begin{center}\small
  \begin{tabular}{lll}
    \toprule
    Comparación & Condición de mejora & Régimen fuerte\\
    \midrule
    Marginal en $a$ & $\eta(b-2a)>1$ & $a<a^*$\\
    Frente a no usar IA & $a>0$, $\eta(b-a)>1$ & $0<a<a_R$\\
    Entre $u$ y $v$ & $\eta(b-u-v)>1$ & $u+v<a_R$\\
    \bottomrule
  \end{tabular}\end{center}
  \source{Las dos últimas equivalencias se refieren a la región activa; fuera de ella se aplica la saturación.}
\end{frame}

\begin{frame}{Corolario 1: qué determina los umbrales y las ganancias}
  En el régimen fuerte, $\eta_c=1/b$ y
  \[
    \frac{\partial a^*}{\partial b}=\frac12,\quad
    \frac{\partial a_R}{\partial b}=1,\qquad
    \frac{\partial a^*}{\partial\eta}=\frac1{2\eta^2},\quad
    \frac{\partial a_R}{\partial\eta}=\frac1{\eta^2}.
  \]
  Un mayor esfuerzo inicial $b$ reduce la complementariedad crítica;
  una mayor $\eta$ amplía la región de mejora. Las otras tasas no mueven los umbrales.
  \par\medskip
  La ganancia máxima de aprendizaje es
  \[
    \Delta L=L(a^*)-L(0)=\frac{\kappa(\eta b-1)^2}{4\eta},\qquad
    \pi_H(a^*)-\pi_H(0)=
    \frac{\mu\,\Delta L}{[L(a^*)+\mu][L(0)+\mu]}.
  \]
  Cerca de $\eta b=1$, la mejora es de segundo orden. El denominador explica
  por qué mayor aprendizaje adicional no se traduce proporcionalmente en habilidad.
\end{frame}

\begin{frame}{Preguntas para la discusión}
  \begin{enumerate}
    \item ¿Es convincente la complementariedad en el aprendizaje cuando el
    trabajador sigue siendo miope, o conviene que el siguiente modelo
    incorpore decisiones de esfuerzo con visión de futuro?
    \par\medskip
    \item ¿Qué resultado observable permitiría distinguir $\eta$ del aprendizaje
    basal $\lambda_0$ y de la eficacia del esfuerzo $\kappa$?
    \par\medskip
    \item ¿Qué condiciones sobre transiciones adicionales permitirían conservar
    la reversión en tres estados, y qué resultado de bienestar debería acompañarla?
  \end{enumerate}
  \par\medskip
  El resultado demostrado pertenece a dos estados y a asistencia fija.
  La extensión a más estados y el diseño de políticas quedan abiertos.
